ภาษา C# พื้นฐาน

ภาษาสุดคลาสสิกจาก Microsoft ผู้อยู่เบื้องหลังเกม Unity และแอป Windows

1. C# (C-Sharp) คืออะไร?

C# เป็นภาษาที่พัฒนาโดย Microsoft มีความทันสมัยและจัดการหน่วยความจำอัตโนมัติ (Garbage Collection) นิยมใช้พัฒนาเกมด้วย Unity, พัฒนาแอปพลิเคชัน Windows และระบบหลังบ้านด้วย .NET Framework

2. โครงสร้างโปรแกรมแรก

using System;

class Program {
    static void Main() {
        // พิมพ์ข้อความออกหน้าจอ
        Console.WriteLine("Hello, World!");
    }
}

3. ตัวแปรและ String Interpolation

C# มีลูกเล่นการแทรกตัวแปรลงในข้อความด้วยเครื่องหมาย $ ที่อ่านง่ายมาก

string name = "Mate";
int age = 25;

// ใช้ $ นำหน้าข้อความ แล้วใส่ตัวแปรใน {}
Console.WriteLine($"สวัสดี {name} อายุ {age} ปี");

4. การสร้างฟังก์ชัน (Methods)

static int AddNumbers(int a, int b) {
    return a + b;
}

static void Main() {
    int result = AddNumbers(5, 10);
    Console.WriteLine("ผลบวกคือ: " + result);
}

* ภาษา C# ต้องใช้ .NET SDK และโปรแกรมอย่าง Visual Studio ในการรัน
